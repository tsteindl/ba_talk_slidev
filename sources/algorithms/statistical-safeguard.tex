\begin{algorithm}
    \caption{Statistical safeguard subroutine}
    \label{alg:stat-safeguard-subroutine}
    \begin{algorithmic}
        \Procedure{StatisticalSafeguard}
        {$B, N_{\text{min}}, N_{\text{max}},
                \epsilon, \hat\phi_p, \sigma_p^2,
                \phi_{\text{min}}, \phi_{\text{max}}$}
        \State $N^* \gets N_\text{min},
            \qquad P^* \gets 0$

        \For{$N' = N_\text{min}, \ldots, N_{\text{max}}$}
        \State $m \gets \left\lfloor B / N' \right\rfloor$
        \State $\sigma \gets \dfrac{1}{2N'\sqrt{m}}$

        \State $P_{\text{valid}}
            \gets
            F_{\hat\phi_p,\sigma_p^2;
            [\phi_{\text{min}},\phi_{\text{max}}]}
            \left(\dfrac{\pi}{2N'}\right)$

        \State $P_{\text{acc}}
            \gets
            1 - 2\Phi\left(-\dfrac{\epsilon}{\sigma}\right)$

        \State $P \gets P_{\text{valid}} \cdot P_{\text{acc}}$

        \If{$P > P^*$}
        \State $P^* \gets P,
            \qquad N^* \gets N'$
        \EndIf
        \EndFor

        \State \Return $N^*$
        \EndProcedure
    \end{algorithmic}
\end{algorithm}
