\begin{algorithm}
    \caption{Simulate $\phi$ using Binary Search}
    \label{alg:binary-search}
    \begin{algorithmic}
        \Procedure{SimulatePhiBinarySearch}{\textit{budget}, $\phi_\text{min}, \phi_\text{max}, \textit{m}', \alpha, \epsilon$}
        \State $N_\text{min} \gets \left\lfloor\frac{\pi}{2 \phi_\text{max}}\right\rfloor,
            \qquad N_\text{max} \gets \left\lfloor\frac{\pi}{2 \phi_\text{min}}\right\rfloor$
        \State $N\gets N_\text{min}$

        \State $\hat \phi \gets \Call{Simulate}{N, m'},
            \qquad B_\text{used} \gets m' N$

        \State $\hat\phi_\text{acc} \gets \hat\phi,\qquad N_\text{acc} \gets N$ \Comment{Pilot: deepest probe not classified as an overshoot}

        \State Initialize lower threshold $\phi_1$ such that $\mathbb{P}\!\left(\tilde\phi_N \leq \phi_1 \mid \hat\phi_\text{acc}\right)=\alpha$, using $\tilde\phi_N \mid \hat\phi_\text{acc} \sim \mathcal{N}\!\left(\hat\phi_\text{acc}, \tfrac{1}{4m'N^2}\right)$
        \State

        \State $L \gets N_\text{min},\qquad U \gets N_\text{max}$ \Comment{Lower and upper bound for $N$}
        \State $N \gets N + \lfloor (U-N)/2 \rfloor$ \Comment{Update $N$ to midpoint of the set $\{N,\ldots, U\}$}

        \Do
        \State \textbf{if} the next probe does not fit in the remaining budget \textbf{then break}
        \State $\hat \phi \gets \Call{Simulate}{N, \textit{m}'},
            \qquad B_\text{used} \gets B_\text{used} + m' N$

        %    \State Update budget used

        \State $N_\text{old} \gets N$ \Comment{Temporary $N$}

        \If {$\hat \phi < \phi_1$}
        \State $N \gets N - \lfloor (N - L)/2 \rfloor$ \Comment{Update $N$ to midpoint of the set $\{L,\ldots, N\}$}
        \State $U \gets N_\text{old}$ \Comment{Update upper bound for $N$}
        \Else
        \State $\hat\phi_\text{acc} \gets \hat\phi, \qquad N_\text{acc} \gets N_\text{old}$ \Comment{Sharper pilot}
        \State $N \gets N + \lfloor (U - N) / 2 \rfloor$ \Comment{Update $N$ to midpoint of the set $\{N, \ldots, U\}$}
        \State $L \gets N_\text{old}$ \Comment{Update lower bound for $N$}
                \State
                \State Recompute $\phi_1$ such that $\mathbb{P}\!\left(\tilde\phi_N \leq \phi_1 \mid \hat\phi_\text{acc}\right)=\alpha$, using $\tilde\phi_N \mid \hat\phi_\text{acc} \sim \mathcal{N}\!\left(\hat\phi_\text{acc}, \tfrac{1}{4m'N^2}\right)$
        \EndIf

        \doWhile {$N_\text{old} \neq N$}

        \State
        \State $\sigma_\text{acc}^2 \gets \frac{1}{4m'N_\text{acc}^2}$
        \State $B_\text{rem} \gets \textit{budget} - B_\text{used}$
        \State $N^* \gets \Call{StatisticalSafeguard}{B_\text{rem}, N_\text{min}, N_\text{max}, \epsilon, \hat\phi_\text{acc}, \sigma_\text{acc}^2, \phi_\text{min}, \phi_\text{max}}$

        \State $m \gets \lfloor B_\text{rem} / N^*\rfloor$
        \Comment{Calculate shots such that remaining budget is exhausted}
        \State $\hat \phi \gets \Call{Simulate}{N^*, \textit{m}}$

        \State \Return $\hat\phi$

        \EndProcedure
    \end{algorithmic}
\end{algorithm}
